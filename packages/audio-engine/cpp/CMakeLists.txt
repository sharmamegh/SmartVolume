cmake_minimum_required(VERSION 3.20)
project(smartvolume_dsp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(smartvolume_dsp STATIC src/Meter.cpp)
target_include_directories(smartvolume_dsp PUBLIC include)
enable_testing()
add_executable(meter_test tests/MeterTest.cpp)
target_link_libraries(meter_test PRIVATE smartvolume_dsp)
add_test(NAME meter_test COMMAND meter_test)
